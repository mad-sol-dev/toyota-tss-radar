# SocketCAN support (candump/cansend, ip link) comes from the system

# DBC decoding of the Toyota ADAS frames (cantools.database namespace,
# not the deprecated cantools.db).
cantools>=39

# Only for radarmap.py (bird's-eye view of the tracks) and make_test_log.py.
# canwatch.py / canlog_stats.py / spoof_dsu.py run without these three.
matplotlib>=3.5
numpy>=1.21
Pillow>=9        # PillowWriter for the GIF export in radarmap.py --gif
